# -*- coding: utf-8 -*-
import os, json, traceback

import clr
clr.AddReference('System')
from System.Net import WebRequest, SecurityProtocolType, ServicePointManager, WebException
from System.IO import StreamReader
from System.Text import Encoding

# TLS 1.2 на всякий случай
ServicePointManager.SecurityProtocol = SecurityProtocolType.Tls12

# -------- настройки --------
OPENROUTER_MODELS = [
    "openai/gpt-oss-120b:free",
    # при желании добавьте резервные модели
    # "google/gemma-2-9b-it:free",
    # "meta-llama/llama-3-8b-instruct:free"
]
OPENROUTER_REFERER = "https://github.com/kraer/AiRevit"  # можно оставить
OPENROUTER_X_TITLE = "AiRevit Plugin"

SYSTEM_PROMPT = """
Ты — Revit BIM ассистент. Твоя задача — анализировать промпты пользователя и извлекать из них параметры для создания архитектурных и инженерных элементов (в частности, стен).
Ты ВСЕГДА должен возвращать ответ в формате строгого JSON без дополнительных пояснений или Markdown разметки.

Формат JSON:
{
  "element_type": "string (например, 'wall')",
  "wall_type_keyword": "string (ключевое слово типа стены, например 'Гипсокартон')",
  "length_mm": number (длина в миллиметрах),
  "start_point": [x, y, z] (массив чисел, обычно [0,0,0] если не указано иное),
  "level_name": "string (имя уровня, по умолчанию 'Уровень 1')"
}
"""

def _try_openrouter(messages, model_name, api_key):
    url = "https://openrouter.ai/api/v1/chat/completions"
    req = WebRequest.Create(url)
    req.Method = "POST"
    req.ContentType = "application/json"

    # Заголовки
    try:
        req.Headers.Add("Authorization", "Bearer " + api_key)
    except Exception:
        # На некоторых .NET возможны ограничения заголовков — попробуем ещё раз
        req.Headers["Authorization"] = "Bearer " + api_key
    if OPENROUTER_X_TITLE:
        req.Headers.Add("X-Title", OPENROUTER_X_TITLE)
    if OPENROUTER_REFERER:
        # OpenRouter принимает HTTP-Referer, а .NET ещё и Referer-свойство
        try:
            req.Headers.Add("HTTP-Referer", OPENROUTER_REFERER)
        except Exception:
            pass
        try:
            req.Referer = OPENROUTER_REFERER
        except Exception:
            pass

    payload = {
        "model": model_name,
        "messages": messages,
        "temperature": 0.0
    }
    data = json.dumps(payload)
    bytes_ = Encoding.UTF8.GetBytes(data)
    req.ContentLength = bytes_.Length

    # Пишем тело
    s = req.GetRequestStream()
    s.Write(bytes_, 0, bytes_.Length)
    s.Close()

    try:
        # Ответ
        resp = req.GetResponse()
        r = StreamReader(resp.GetResponseStream(), Encoding.UTF8).ReadToEnd()
        resp.Close()

        raw = json.loads(r)
        if "choices" in raw and raw["choices"]:
            content = raw["choices"][0]["message"]["content"]
            content = (content or "").strip()
            # Срезаем ```json ... ```
            if content.startswith("```"):
                content = content.strip("`").strip()
                if content.lower().startswith("json"):
                    content = content[4:].strip()
            parsed = json.loads(content)
            return {"ok": True, "data": parsed, "raw": raw}
        else:
            return {"ok": False, "error": "EmptyChoices", "body": raw}
    except WebException as e:
        body = None
        if e.Response is not None:
            body = StreamReader(e.Response.GetResponseStream(), Encoding.UTF8).ReadToEnd()
        return {"ok": False, "error": "WebException", "message": str(e), "body": body}
    except Exception as e:
        return {"ok": False, "error": "Exception", "message": str(e), "trace": traceback.format_exc()}

def get_bim_parameters_from_text(user_input, api_key):
    msgs = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user",   "content": user_input}
    ]
    for m in OPENROUTER_MODELS:
        r = _try_openrouter(msgs, m, api_key)
        if r and r.get("ok"):
            return r
    return r  # вернём последнюю ошибку/ответ

# -------- входы из Dynamo --------
# IN[0] — prompt (строка); IN[1] — (опционально) apiKey
prompt = None
api_key = None

try:
    prompt = IN[0] if len(IN) > 0 else None
    api_key = IN[1] if len(IN) > 1 and IN[1] else None
except Exception:
    pass

# Подстрахуем значения
if not prompt:
    prompt = "Смоделируй гипсокартонную стену длиной 1000 миллиметров."
if not api_key:
    api_key = os.environ.get("OPENROUTER_API_KEY")

if not api_key:
    OUT = {"ok": False, "error": "NoApiKey", "hint": "Передайте ключ во второй вход или задайте переменную среды OPENROUTER_API_KEY и перезапустите Revit"}
else:
    OUT = get_bim_parameters_from_text(prompt, api_key)

В Watch вы увидите словарь:
Если ok: True — в поле data будет ваш JSON с полями element_type, length_mm и т.д.
Если ok: False — будет error/message/body/trace с понятной причиной.
Типичные ошибки и что сделать

NoApiKey — передайте ключ во второй вход apiKey или задайте переменную среды и перезапустите Revit.
WebException с 401 — ключ неверный.
WebException с 403 и текстом про Referer/X-Title — оставьте X-Title и Referer как в коде (уже сделано).
WebException The remote name could not be resolved / таймаут — нет выхода в интернет/DNS, проверьте прокси.
Could not create SSL/TLS secure channel — обновите Revit/Dynamo или оставьте строку с TLS 1.2 (уже в коде).
Exception JSONDecodeError — модель вернула нестрогий JSON. Мы уже пытаемся срезать json … . Если нужно — добавим дополнительную очистку.
Пара мелочей

В CPython ноде (Python Script 3) import clr и .NET работают — всё ок.
Если у вас только «Python Script» (старый IronPython 2.7) — код тоже будет работать.
Если IN[0] не подключён — раньше у вас улетало в API null. В моём коде стоит дефолтный prompt, чтобы не падало.
